cmake_minimum_required(VERSION 3.20)
project(sage_einsums_cpp LANGUAGES CXX)
option(SAGE_BUILD_FULL_API "Build the CPU/Python library (Eigen, JSON, HDF5)" ON)
add_executable(einsums-cpp-candidate driver.cpp)
target_compile_features(einsums-cpp-candidate PRIVATE cxx_std_20)
target_compile_options(einsums-cpp-candidate PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
if(SAGE_BUILD_FULL_API)
    find_package(Eigen3 REQUIRED NO_MODULE)
    if(Eigen3_VERSION VERSION_LESS 3.4)
        message(FATAL_ERROR "Eigen 3.4 or newer is required")
    endif()
    find_package(nlohmann_json 3.11 REQUIRED)
    # HDF5's C API is used through a C++ RAII boundary.
    enable_language(C)
    find_package(HDF5 REQUIRED COMPONENTS C)
    add_library(sage_einsums_cpp SHARED api/execute.cpp api/linalg.cpp api/storage.cpp)
    target_compile_features(sage_einsums_cpp PUBLIC cxx_std_20)
    target_include_directories(sage_einsums_cpp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(sage_einsums_cpp PUBLIC Eigen3::Eigen nlohmann_json::nlohmann_json hdf5::hdf5)
    target_compile_options(sage_einsums_cpp PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    add_executable(einsums-cpp-library-candidate library_driver.cpp)
    target_link_libraries(einsums-cpp-library-candidate PRIVATE sage_einsums_cpp)
    target_compile_options(einsums-cpp-library-candidate PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    install(TARGETS sage_einsums_cpp LIBRARY DESTINATION lib)
    install(DIRECTORY api/ DESTINATION include/sage-einsums/api FILES_MATCHING PATTERN "*.hpp")
    install(FILES tensor.hpp kernels.hpp sage_api.h DESTINATION include/sage-einsums)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(einsums-cpp-tests tests.cpp)
    target_compile_features(einsums-cpp-tests PRIVATE cxx_std_20)
    target_compile_options(einsums-cpp-tests PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    # Keep contract assertions active even in Release builds.
    target_compile_options(einsums-cpp-tests PRIVATE -UNDEBUG)
    add_test(NAME tensor-contract COMMAND einsums-cpp-tests)
    if(SAGE_BUILD_FULL_API)
        add_executable(einsums-cpp-api-tests api_tests.cpp)
        target_link_libraries(einsums-cpp-api-tests PRIVATE sage_einsums_cpp)
        target_compile_options(einsums-cpp-api-tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        add_test(NAME cpu-api-contract COMMAND einsums-cpp-api-tests)
    endif()
endif()
